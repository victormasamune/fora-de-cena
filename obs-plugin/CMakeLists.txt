cmake_minimum_required(VERSION 3.20)
project(fora_de_cena_obs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(libobs REQUIRED)
add_library(fora_de_cena_obs MODULE src/plugin-main.cpp)
target_link_libraries(fora_de_cena_obs PRIVATE OBS::libobs)
target_compile_definitions(fora_de_cena_obs PRIVATE PLUGIN_VERSION="\${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(fora_de_cena_obs PRIVATE /W4 /permissive-)
else()
  target_compile_options(fora_de_cena_obs PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(fora_de_cena_obs PROPERTIES PREFIX "" OUTPUT_NAME "fora-de-cena-obs")
